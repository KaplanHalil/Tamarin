  exists-trace
  "∃ mn1 kn sak1 mn3 kn1 sak2 #i #j.
    (InitialSessionCompleted( mn1, kn, sak1 ) @ #i) ∧
    (RekeySessionCompleted( mn3, kn1, sak2 ) @ #j)"
/*
guarded formula characterizing all satisfying traces:
"∃ mn1 kn sak1 mn3 kn1 sak2 #i #j.
  (InitialSessionCompleted( mn1, kn, sak1 ) @ #i) ∧
  (RekeySessionCompleted( mn3, kn1, sak2 ) @ #j)"
*/
simplify
solve( !SharedCAK( cak ) ▶₁ #i )
  case Setup_CAK
  solve( KSSTATE0( mn, kn, sak1 ) ▶₂ #i )
    case Initial_KeyServer_Send
    solve( !SharedCAK( cak.2 ) ▶₁ #j )
      case Setup_CAK
      solve( KSSTATE2( mn2, kn1 ) ▶₂ #j )
        case Rekey_KeyServer_Send
        solve( !KU( count(count(count(~mn.1))) ) @ #vk.10 )
          case Rekey_Server_Receive
          solve( !KU( count(~mn) ) @ #vk.4 )
